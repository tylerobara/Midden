  http://{{ (first .Values.ingress.hosts).host | default (printf "%s.%s.svc" (include "midden.fullname" .) .Release.Namespace) }}

Sign in as {{ .Values.secret.adminUser | default "admin" }}. If you did not set

  kubectl -n {{ .Release.Namespace }} logs deploy/{{ include "midden.fullname" . }} | grep 'one-time password'
